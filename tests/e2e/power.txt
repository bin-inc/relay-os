# reboot and poweroff (spec §7.4): the machine restarts through the FADT
# reset register and boots again on the same disk, then test mode's
# poweroff makes QEMU exit through isa-debug-exit.
timeout 30
expect root@relay:/# $
send echo before
expect \nbefore\n
reboot
expect Relay OS \d+\.\d+\.\d+
expect root@relay:/# $
send echo after
expect \nafter\n
poweroff
